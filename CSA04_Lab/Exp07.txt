#include <stdio.h>

int main() {
    int n, i, time = 0, done = 0, idx;
    int at[20], bt[20], ct[20], wt[20], tat[20];
    int completed[20] = {0};

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        printf("Enter arrival time and burst time for P%d: ", i + 1);
        scanf("%d%d", &at[i], &bt[i]);
    }

    while (done < n) {
        idx = -1;

        for (i = 0; i < n; i++) {
            if (!completed[i] && at[i] <= time) {
                if (idx == -1 || bt[i] < bt[idx])
                    idx = i;
            }
        }

        if (idx == -1) {
            time++;
            continue;
        }

        time += bt[idx];
        ct[idx] = time;
        completed[idx] = 1;
        done++;
    }

    printf("\nP\tAT\tBT\tWT\tTAT\n");
    for (i = 0; i < n; i++) {
        tat[i] = ct[i] - at[i];
        wt[i] = tat[i] - bt[i];
        printf("P%d\t%d\t%d\t%d\t%d\n",
               i + 1, at[i], bt[i], wt[i], tat[i]);
    }

    return 0;
}
