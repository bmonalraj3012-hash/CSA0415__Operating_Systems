#include <stdio.h>
#include <pthread.h>

void *display(void *arg) {
    int id = *(int *)arg;
    printf("Thread %d is running\n", id);
    return NULL;
}

int main() {
    pthread_t t1, t2;
    int a = 1, b = 2;

    if (pthread_create(&t1, NULL, display, &a) != 0) {
        printf("Thread creation failed\n");
        return 1;
    }

    if (pthread_create(&t2, NULL, display, &b) != 0) {
        printf("Thread creation failed\n");
        pthread_join(t1, NULL);
        return 1;
    }

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    printf("Main thread completed\n");
    return 0;
}
